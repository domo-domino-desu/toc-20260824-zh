  \textbf{记号一览}
    \\[1.5ex]
\begin{tabular}{r@{\hspace*{2em}}l}
  \multicolumn{1}{c}{\tabulartext{记号}}
    &\multicolumn{1}{c}{\tabulartext{说明}}          \\
  \hline
  $\powerset(S)$ &$S$ 的幂集，即 $S$ 的所有子集组成的集合 \\
  $\setcomp{S}$  &集合~$S$ 的补集 \\
  $\charfcn{S}$  &集合~$S$ 的特征函数 \\
  $\sequence{a_0,a_1, \ldots{}}$  &序列 \\
  $\N$, $\Z$, $\Q$, $\R$  &自然数 $\set{0, 1, \ldots{}}$、整数、有理数、实数 \\
  \str{a}, \str{b}, \ldots{} \str{0}, \str{1}  &字符（注意字体） \\
  $\Sigma$ &字母表（字符的集合） \\
  $\B$     &位字符的字母表 $\set{\str{0},\str{1}}$，或位集合 $\set{0,1}$ \\
  $\sigma$, $\tau$ &串（除 $\TMfcn$ 和 $\emptystring$ 外的任意小写希腊字母） \\
  $\emptystring$ &空串 \\
  $\kleenestar{\Sigma}$ &该字母表上所有串的集合 \\
  $\lang$  &语言，即 $\kleenestar{\Sigma}$ 的子集 \\
  $\TM$    &Turing 机 \\
  $\TMfcn$ &Turing 机计算的函数 \\
  $\TMfcn(x)\converges$, $\TMfcn(x)\diverges$ &函数在该输入上收敛或发散 \\
  % $\UTM$   &universal Turing machine                                  \\
  $\mathcal{G}$  &图 \\
  $\FSM$   &有限状态机 \\
  $\bigOh(f)$    &函数的增长阶 \\
  $\compclass{Class}$    &复杂度类所用字体 \\
  $\P$, $\NP$  &多项式时间类与非确定性多项式时间类 \\
  $\probname{Problem}$   &问题名称所用字体 \\
  $\TM[V]$ &$\NP$ 语言的验证器 \\
\end{tabular}
\end{center}
