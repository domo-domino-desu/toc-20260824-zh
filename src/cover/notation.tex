\begin{center}\small
  \textbf{Notation summary}
    \\[1.5ex]
\begin{tabular}{r@{\hspace*{2em}}l}
  \multicolumn{1}{c}{\tabulartext{Notation}}
    &\multicolumn{1}{c}{\tabulartext{Description}}          \\
  \hline
  $\powerset(S)$ &power set, collection of all subsets of $S$ \\
  $\setcomp{S}$  &complement of the set~$S$          \\
  $\charfcn{S}$  &characteristic function of the set~$S$  \\
  $\sequence{a_0,a_1, \ldots{}}$  &sequence  \\
  $\N$, $\Z$, $\Q$, $\R$  &natural numbers $\set{0, 1, \ldots{}}$,
                           integers, rationals, reals \\ 
  \str{a}, \str{b}, \ldots{} \str{0}, \str{1}  & character (note the typeface)        \\  
  $\Sigma$ &alphabet, set of characters   \\
  $\B$     &alphabet of bits characters $\set{\str{0},\str{1}}$, or set of bits $\set{0,1}$  \\ 
  $\sigma$, $\tau$ &strings (any lower-case Greek letter except $\TMfcn$) \\
  $\emptystring$ &empty string  \\ 
  $\kleenestar{\Sigma}$ &set of all strings over the alphabet  \\
  $\lang$  &language, a subset of $\kleenestar{\Sigma}$   \\
  $\TM$    &Turing machine \\
  $\TMfcn$ &function computed by a Turing machine \\
  $\TMfcn(x)\converges$, $\TMfcn(x)\diverges$ &function converges on that input, or diverges \\
  % $\UTM$   &universal Turing machine                                  \\
  $\mathcal{G}$  &graph                                         \\
  $\FSM$   &Finite State machine \\
  $\bigOh(f)$    &order of growth of the function \\
  $\compclass{Class}$    &complexity class font  \\
  $\P$, $\NP$  &polynomial time and nondeterministic polynomial time classes \\
  $\probname{Problem}$   &problem name font  \\
  $\TM[V]$ &verifier for an $\NP$ language \\
\end{tabular}
\end{center}
