\question
Consider these two graphs, $\mathcal{G}_0$ and $\mathcal{G}_1$.
\begin{center}
  \includegraphics{asy/languages002.pdf}
  \qquad
  \includegraphics{asy/languages003.pdf}
\end{center}
\begin{parts}
\part List the vertices of $\mathcal{G}_1$.
\part List the vertices of $\mathcal{G}_2$.
\part Give the edges of $\mathcal{G}_1$.
\part Give the edges of $\mathcal{G}_2$.
\part Give the degree sequence of $\mathcal{G}_1$.
\part Give the degree sequence of $\mathcal{G}_2$.
\part Consider this correspondence between the vertices.
\begin{center}
\begin{tabular}{r|*{8}{c}}
  vertex of $\mathcal{G}_1$ &$v_0$ &$v_1$ &$v_2$ &$v_3$ &$v_4$ &$v_5$ &$v_6$ &$v_7$ \\
  \cline{2-9}
  vertex of $\mathcal{G}_2$ &$v_6$ &$v_2$ &$v_7$ &$v_3$ &$v_5$ &$v_1$ &$v_0$ &$v_4$ 
\end{tabular}
\end{center}
Find the image, under the correspondence of the edges of $\mathcal{G}_1$.
Do they correspond to the edges of $\mathcal{G}_2$?
\part Argue that the two are not isomorphic.
\end{parts}
\end{questions}
