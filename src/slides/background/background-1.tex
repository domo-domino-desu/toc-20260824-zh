\begin{example}
These sets have the same cardinality:
(a)~$|A|=|B|$ where 
$A=\set{x\in\R\suchthat 0\leq x<1}$
and $B=\set{x\in\R\suchthat 0\leq x<10}$
(b)~$|C|=|D|$ where 
$C=\set{x\in\R\suchthat 0\leq x<1}$
and $D=\set{x\in\R\suchthat 1\leq x<5}$
(c)~$|E|=|F|$ where 
$E=\set{x\in\N\suchthat 0\leq x<10}$
and $D=\set{x\in\N\suchthat 1\leq x<11}$
\end{example}

\begin{frame}
  \frametitle{Cross product}
\begin{example}
If $N_2=\set{0,1}\times\N$ then $|N_2|=|\N|$.
If $N_3=\set{0,1,2}\times\N$ then $|N_3|=|\N|$.
For any finite~$k$, 
if $N_k=\set{0,1,\ldots k-1}\times\N$ then $|N_k|=|\N|$.
\end{example}
