\begin{frame}
  \frametitle{Definitions}
We use these to compare how algorithms consume computing resources.
  
\ExecuteMetaData[\catchfilefn]{def:ComplexityFcn}

\fbox{%
  \alert{Think of `$f$ is $\bigOh(g)$' as meaning that
    $f$'s order of growth is less than or equal to $g$'s.}
  }

\pause\medskip
\noindent\textcolor{blue}{\large Points:}
\begin{itemize}
\item Turing machines are discrete devices so we might think that
  functions $\map{f}{\N}{\N}$ are suitable.
  But experience shows that  
      functions with real number inputs and outputs are best. 
      (The term `complexity function' is not standard
      but we will find it convenient.)
    \item
      We use the letter `$\bigOh$' because this is about the order of growth.
    \item
      The `$f=\bigOh(g)$' notation is awkward but 
      standard.
      The notation `$f\in\bigOh(g)$' is better but not as common.
\end{itemize}
\end{frame}


\begin{frame}{$\bigOh$ compares orders of growth}
For both of these $f$ is $\bigOh(g)$, that is,
$f$'s growth rate is less than or equal to $g$'s rate.
\begin{center}
   % \vcenteredhbox{\includegraphics{asy/complexity00.pdf}}
   % \hspace{2em}

   \includegraphics{asy/complexity01.pdf}
   \hspace{2em}
   \includegraphics{asy/complexity00.pdf}
\end{center}


\begin{itemize}
\item
On the left, $f$ is $\bigOh(g)$ despite that $g(x)<f(x)$ for small~$x$'s.
As $x$ goes to infinity, $g$ races ahead of~$f\!$.
So $f$'s order of growth is strictly less than $g$'s.

\item 
On the right, $f$ is $\bigOh(g)$ despite that $g(x)< f(x)$ for all $x>0$.
As $x$ goes to infinity the two functions track together, so 
$f$'s order of growth is less than or equal to $g$'s,
since it is equal.
\end{itemize}
\end{frame}

\begin{frame}{Computing Big-$\bigOh$}

For many familiar functions the next result is a convenient way to compute
orders of growth.

\ExecuteMetaData[\catchfilefn]{lem:LHBigO}

\pause
To compute the limits we often apply this theorem from 
Calculus.

\ExecuteMetaData[\catchfilefn]{th:LHopital}

\begin{example}
Consider $f(x)=3x^2-2x+4$ and $g(x)=(1/2)x^5-6$.
Take derivatives twice and apply the theorem above 
to conclude that $f$ is~$\bigOh(g)$
but $g$ is not~$\bigOh(f)$.
\begin{equation*}
  \lim_{x\to\infty}\frac{f(x)}{g(x)}
  =
  \lim_{x\to\infty}\frac{3x^2-2x+4}{(1/2)x^5-6}
  =
  \lim_{x\to\infty}\frac{6x-2}{(5/2)x^4}
  =
  \lim_{x\to\infty}\frac{6}{10x^3}=0
\end{equation*}
\end{example}
\end{frame}


\begin{frame}
\begin{example}
  The polynomial function $f(n)=5n^3$ has a smaller order of growth than
  the exponential function
  $g(n)=3\cdot 2^n+4$.
  Here we do the limit with L'H\^{o}pital's Rule.
  \begin{equation*}
    \lim_{x\to\infty}\frac{5x^3}{3\cdot 2^x+4}
    =\lim_{x\to\infty}\frac{15x^2}{3\ln(2)\cdot 2^x}
    =\lim_{x\to\infty}\frac{30x}{3(\ln(2))^2\cdot 2^x}
    =\lim_{x\to\infty}\frac{30}{3(\ln(2))^3\cdot 2^x}
    =0
  \end{equation*}
\end{example}

\begin{example}
  The function  $f(n)=5n^3$
  has a larger order of growth than the logarithmic function
  $g(n)=7\lg(n)$.
  \begin{equation*}
    \lim_{x\to\infty}\frac{5x^3}{7\lg x}
    =\lim_{x\to\infty}\frac{15x^2}{7(1/x)}
    =\lim_{x\to\infty}\frac{15x^3}{7}
    =\infty
  \end{equation*}
\end{example}
\end{frame}

\begin{frame}{The Order of Growth Hierarchy}
\ExecuteMetaData[\catchfilefn]{lem:PolySlowerThanExp}

\begin{example}
Where $g$~is the polynomial function $g(x)=x^2$,
this shows that $f(x)=\lg(x)$ is $\bigOh(g)$.
\begin{equation*}
  \lim_{x\to\infty}\frac{f(x)}{g(x)}
  =
  \lim_{x\to\infty}\frac{1/(x\cdot \ln(2))}{2x}
  =\lim_{x\to\infty}\frac{1}{x^2\cdot 2\ln(2)}
  =0
\end{equation*}
\end{example}

\begin{example}
This shows that the polynomial $g(x)=x^2$ is $\bigOh(h)$ where
$h$~is the exponential $h(x)=2^x\!$.
\begin{equation*}
  \lim_{x\to\infty}\frac{g(x)}{h(x)}
  =
  \lim_{x\to\infty}\frac{x^2}{2^x}
  =
  \lim_{x\to\infty}\frac{2x}{2^x\cdot \ln(2)}
  =
  \lim_{x\to\infty}\frac{2}{2^x\cdot \ln(2)^2}
  =0
\end{equation*}
\end{example}
\end{frame}

\alert{Cobham's thesis} is that 
the \alert{tractable}

\begin{frame}{Representations}
We shall assume that for our
algorithms all of the inputs and outputs are in some reasonably 
efficient bitstring representation.

Thus, we shall understand ``given two numbers $n_0,n_1\in\N$'' to mean
`given the bitstring representation of two numbers'.
And 
we shall take ``let $\mathcal{G}$ be a graph'' to mean
`given a reasonably efficient representation of a graph~$\mathcal{G}$'.

Having noted it, we will from now on generally ignore it.
\end{frame}


\begin{frame}{Types of problems}
\begin{itemize}
\item A \alert{function problem}
   asks that the algorithm have a single output for each input.
\pause
\item A \alert{search problem} asks for an algorithm that, 
  while there may be many
  solutions in the search space, will stop when it has found any one.
\pause
\item An \alert{optimization problem} asks for an algorithm that
  gives a solution that is  
  best according to some metric.
\pause
\item A \definend{decision problem} 
  asks for an algorithm with a `Yes' or `No' answer.
  (So it is a Boolean function.
  But is important enough to list on its own, instead of subsumed
  as a function problem.)
\pause
\item Often a decision problem is expressed as a
   \alert{language recognition problem},
   where we are given some language and asked for an algorithm to
   decide if its input is a member of that language.
\end{itemize}

The last is the one that we will work with most often.
\end{frame}



\begin{frame}{Problems}
In this chapter, \alert{problem} usually
means a decision problem for a language.
If we get a problem that isn't a language decision problem
then we will often recast it as one.

\begin{example}
\probname{Subset Sum} inputs a set~$S$ of numbers 
and a target number~$T$, and then decides 
whether a subset of~$S$ adds to the target.
We can recast it as the problem of deciding membership in this language.
\begin{equation*}
  \lang=\set{\sequence{S,T}\suchthat \text{A subset of $S$ adds to~$T$}}
\end{equation*}
Restating the problem as this language leaves it essentially unchanged.
Instead of asserting `there is a subset of $S=\set{19,21,52,106}$ that adds
to~$T=40$' we may say `$\sequence{S,T}\in\lang$'. 
\end{example}

\begin{example}
Consider \probname{Vertex-to-Vertex Path}, where the input is 
a graph and two vertices in the graph and the problem is to decide if
there is a path between the two.
We recast it as the  decision problem for this language.
\begin{equation*}
  \lang=\set{\sequence{\mathcal{G},v,\hat{v}}\suchthat
    \text{there is a path in~$\mathcal{G}$ between the vertices $v$ and~$\hat{v}$}}
\end{equation*}
\end{example}

\begin{example}
To recast \probname{Shortest Path}, introduce a bound~$B\in\N$ parameter.
\begin{equation*}
  \lang_B=\set{\sequence{\mathcal{G},v,\hat{v}}\suchthat
               \text{there is a path between the vertices of length bounded by~$B$}}
\end{equation*}
To find the length of the shortest path, 
see if there is a $B=1$ path, then see if there
is a $B=2$ path, etc. 
\end{example}
\end{frame}

\begin{frame}{Classes}
\ExecuteMetaData[\catchfilefn]{def:ComplexityClass}

Another example is that we might consider all 
language decision problems for which there is 
an algorithm that runs in space $\bigOh(n)$. 
Still another is the set of  
languages where the decision problem for the language's complement 
has an algorithm that runs in polynomial time. 
\end{frame}

\section{Complexity classes}

\begin{frame}{$\P$}
\ExecuteMetaData[\catchfilefn]{def:P}

Here are some languages whose decision problems are in~$\P$:\begin{parenum}
\item the language of correct sums,
$\lang_1=\set{\sequence{a,b,c}\in\N^3\suchthat a+b=c}$,
\item the language of sorted strings,
$\lang_2=\set{\sigma\in\kleenestar{\set{\str{a},\ldots \str{z}}}\suchthat \text{$\sigma$ is in alphabetical order}}$,
\item the language of correct matrix multiplications,
$\lang_3=\set{\sequence{A,B,C}\suchthat\text{the matrices are such that $AB=C$}}$,
\item the language of primes,
$\lang_4=\set{\str{1}^k\suchthat\text{$k$ is prime}}$,
\item the language of nodes with five neighbors,
$\lang_5=\set{\sequence{\mathcal{G},v}\suchthat\text{vertex~$v$ in $\mathcal{G}$ has exactly five neighbors}}$.  
\end{parenum}

\begin{frame}{Effect of the model of computation}
Different models of computation work at different speeds.
In particular, our experience working with Turing machines 
leads us to perceive that they are slow;
does the machine we standardize on affect how we 
classify problems?

Close analysis shows that
Turing machines are slower than other natural models
by no more than a factor of $n^2$ or $n^3\!$.
(We take a model to be `natural' if it
  was not invented just to be a counterexample to this.)
Thus if we have a problem for which there is a
$\bigOh(n)$ algorithm on
another model then when translated to a Turing machine
algorithm it may be $\bigOh(n^4)$.
Under this scenario, 
a problem that falls in the class~$\P$ with one natural model,
including Turing machines,
also falls there using other natural models.

\pause
\begin{remark}  
  There is a possible counterexample to the contention about there being 
  an at-most polytime delay between natural models of 
  physically-realizable devices. 
  Under the Quantum Computing model
  there are several problems with polytime solutions,
  including integer factorization,
  for which we do not know of any polytime solution in a natural
  non-quantum model.
  Whether quantum computers will ever be practical physical devices is not
  at this moment perfectly clear but scientists and engineers are making
  great progress.
  (Also, conceivably there are polytime solutions for these problems
  on Turing machines.)
\end{remark}
\end{frame}

solution form a very important class in practice.

\item As just mentioned, if a problem is in $\P$ based on any familiar model
including Turing machines, RAM, and Racket programs, then
it is in~$\P$ for all of them.

\item We'd like that if two things, $f$ and~$g$, are easy to 
compute then a simple combination of the two is also easy.

\begin{frame}{Nondeterminism}
Recall from the chapter on Finite State machines 
that a machine is nondeterministic if
from a present configuration and input
it may pass to a next configuration where the number of next states
is zero, or one, or more than one.
This contrasts with a deterministic machine where the number of next 
states is always exactly one.

\ExecuteMetaData[\catchfilefn]{def:NondeterministicTM}
\begin{center}
    \includegraphics{../../complexity/asy/ndtm/ndtm001.pdf}
\end{center}

\begin{frame}
Recall also that we have two mental models of how these devices operate.
In the first, the machine is unboundedly parallel\Dash when it is 
called on to go to multiple states,
it forks a child process for each one.
That is, the machine's computation history is a tree and it
simultaneously computes all branches.

In the second model, the machine guesses which next state to follow,
or is told which next state to follow by a demon, and
then it 
deterministically computes a check of that branch of the computation tree.

\pause
Described in the language of the first model, 
we say that an input string is accepted if one of the 
branches on the tree is an accepting branch,
and it does not accept the
input if no branch ends in an accepting state.
In terms of the second model, this means that the input
is accepted if there is a sequences of guesses that the machine could
make, or a sequence of
branching hints that the demon could produce,
that the machine can verify leads to an accepting state.

These two models are equivalent in that for a given nondeterministic machine
they yield the same collection of accepted strings.
\end{frame}

takes as input a graph and two vertices and the triple is in the language if

\begin{equation*}

              \text{there is a path in $\mathcal{G}$ from $v$ to $\hat{v}$}}
\end{equation*}

For the witness~$\omega$, use the path.
We can write a verifier~$\TM[V]$ that accepts a 
triple~$\sigma=\sequence{\mathcal{G},v,\hat{v}}$ along with a 
witness path~$\omega$, 
and tries to verify that the path lies in the graph between
the two vertices.
If there is such a path then there is a pair $\sequence{\sigma,\omega}$ 
that $\TM[V]$ can accept.
If there is no such path then there is no such pair. 
Clearly verification is fast, polytime.
\end{example}

\begin{example}
\alert{\probname{Linear programming}} optimizes some linear function
$f(x_0,\ldots x_{n-1})=a_0x_0+\cdots+a_{n-1}x_{n-1}$, called the objective function, 
subject to a list~$C$ of constraints of the form
$b_{i,0}x_0+\cdots+b_{i,n-1}x_{n-1}\leq d_i$ and $x_i\geq 0$.
To make this a language decision problem, recast it using a parameter 
lower bound~$B$.
\begin{equation*}
  \lang_B=\set{\sequence{f,C}\suchthat
    \text{some $\sequence{x_0,\ldots x_{n-1}}$ satisfies the constraints~$C$ and~$f(x_0,\ldots x_{n-1})\geq B$}}
\end{equation*}
We can write a verifier~$\TM[V]$ that takes in a
pair~$\sequence{\sigma,\omega}$, where $\sigma$ is a pair $\sequence{f,C}$ and
$\omega$ is a tuple,~$\sequence{x_0,\ldots{}x_{n-1}}$,
and where the verifier checks whether the tuple satisfies the
constraints and also makes the objective function larger than~$B$.
If there is such a sequence then $\TM[V]$ can verify it, in polytime.
If there is no such sequence then $\TM[V]$ can never accept its input.  
\end{example}

in time polynomial in the size of the graph.
\end{example}

\begin{frame}{Reduction between language decision problems}
Recall \probname{Vertex-to-Vertex Path}:~given a graph and two vertices,
decide if there is a path between them.
Recall also \probname{Shortest Path}:~given a weighted graph and two
vertices, find the shortest path between them.

These two are related in that if you can find the shortest path 
then you can find
whether there is a path at all.

\pause
As a run-up toward the definition that makes this relationship precise, 
we first state these as language decision problems.
\begin{align*}
  \lang &=\set{\sequence{\mathcal{G},v_0,v_1}\suchthat
               \text{there is a path from $v_0$ to $v_1$ in $\mathcal{G}$}} \\
  \lang_B &=\set{\sequence{\hat{\mathcal{G}},w_0,w_1}\suchthat
               \text{there is a path from $w_0$ to $w_1$ shorter than length~$B$}}
\end{align*}
(With the second, we are applying the standard bound technique, using 
the parameter~$B\in\N$.
Also, there is a type issue because
\probname{Shortest Path} applies to weighted graphs.
But we can patch up the type issue by taking an unweighted graph to be 
a weighted one where each edge has weight~$1$.)
\end{frame}

\begin{frame}{$\probname{Vertex Cover}\leq_p\probname{Set Cover}$}
\probname{Vertex Cover} inputs a graph 
$\mathcal{G}=\sequence{\mathcal{V},\mathcal{E}}$ and a number $B\in\N^+\!$.
It asks if there are vertices $v_{i_0},v_{i_1},\ldots\,v_{i_{B-1}}$ so that 
each edge contains at least one such vertex.

\begin{example}
Your museum has nine corridors and you must have a guard in a corner of each
corridor.
\begin{center}
  \includegraphics{asy/complexity08.pdf}
\end{center}
You can get away with $B=4$ guards, posted at $v_1$, $v_2$, $v_5$, and~$v_7$.
\end{example}

\pause
\probname{Set Cover} inputs a set~$U$, a number $B\in\N^+\!$,
and a collection $S_0, S_1, \ldots\, S_m$ of subsets of~$U$.
It asks if there is a subcollection whose union is the entire set
$S_{i_0}\cup S_{i_1}\cup \cdots \cup S_{i_{B-1}}=U$. 

\begin{example}
Imagine you are hiring people and have ten skills that you need to cover,
$U=\set{0,1,\ldots\,9}$.
The four applicants list these skills.
\begin{equation*}
  S_0=\set{1,4}\quad
   S_1=\set{2,3,4,5}\quad
   S_2=\set{0,7,9}\quad
   S_3=\set{1,6,8}
\end{equation*}
You can get away with hiring only $B=3$ applicants 
because $S_1\cup S_2\cup S_3=U$.
\end{example}
\end{frame}

\begin{frame}
We will show that
$\probname{Vertex Cover}\leq_p\probname{Set Cover}$ by 
showing how to translate instances of \probname{Vertex Cover} into instances
of \probname{Set Cover}.

First write each as a language decision problem.
\begin{align*}
  \lang_0 &=\set{\sequence{U,\set{S_0,\ldots S_m},B}\suchthat
               \text{some $S_{i_0}, S_{i_1}, \ldots\, S_{i_{B-1}}$ covers $U$}}
  \\
  \lang_1 &=\set{\sequence{\mathcal{G},B}\suchthat
               \text{some set $v_{i_0},v_{i_1},\ldots\,v_{i_{B-1}}$ covers all edges}}      
\end{align*}
We must produce a 
polytime computable function~$f$ that
takes in ~$\sigma=\sequence{\mathcal{G},B}$
and outputs $f(\sigma)=\sequence{U,\set{S_0,\ldots S_m},B}$, such that
$\sigma\in\lang_1$ if and only if $f(\sigma)\in\lang_0$. 

\pause
Given $\sigma=\sequence{\mathcal{G},B}$, 
take $U$ to be the set of~$\mathcal{G}$'s edges.
For each of $\mathcal{G}$'s vertices $v_i\in\mathcal{V}$, let
$
S_{v_i}=\set{e\in\mathcal{E}\suchthat \text{$v_i$ is a vertex in $e$}}
$.
Define $f(\sigma)=\sequence{U,\set{S_0,\ldots S_m},B}$.

Clearly we can compute $f$ in polytime.
Further, for the bound~$B$ there is a collection of sets 
$S_{v_0}\cup S_{v_1}\cup \cdots\, S_{v_{B-1}}$ if and only if
the collection $v_0,v_1,\ldots v_{B-1}$ is a vertex cover of~$\mathcal{G}$.
\begin{example}
For the corridor guards, $\mathcal{G}$, 
the associated sets are $U=\set{a,b,\ldots{} i}$, along with
$S_{v_1}=\set{a,b,d}$, 
and $S_{v_2}=\set{b,c,e}$, \ldots{}
and~$S_{v_7}=\set{f,h,i}$.
\begin{center}
  \vcenteredhbox{\includegraphics{asy/complexity09.pdf}}
\end{center}
The set cover $\set{S_{v_1},S_{v_2},S_{v_5},S_{v_7}}$ gives a vertex cover
$\set{v_1,v_2,v_5,v_7}$.
\end{example}
\end{frame}

\begin{frame}{Reflection: why the question is central}
In high school you had Euclid.
This proves that the base angles of an isosceles triangle are equal. 
\begin{center}  % https://www.gutenberg.org/files/21076/21076-pdf.pdf
  \includegraphics[height=2.2in]{pix/euclid.png}
\end{center}
A proof is a sequence of statements, each justified by reference to 
some prior statement, to a law of logic, or to some axiom.
\end{frame}

\begin{frame}
In a proof we require the all steps be typographic 
in the sense that we can for instance combine two prior statements using
a law of logic to get a third statement,
but a proof cannot take creative leaps.
It must be mechanically checkable.  

This is illustrated by Russell and Whitehead's 
\textit{Principia Mathematica}, from the early 1900's, 
which tried to derive all of mathematics from first principles,
in a way that is completely mechanically checkable.
Each step describes the lines from which it is
derived, so that for instance 
you can mechanically check that the second line follows from the
first by applying the result $51.231$.

This appears on p~362. 
\begin{center}  
  \includegraphics[height=1.25in]{pix/russellwhitehead.png}
\end{center}
% So it at least may appear that doing mathematics from first principles,
% going back and proving all results completely, may be extremely slow.
\end{frame}

Then do all two-step combinations, etc.
This process generates all possible theorems, eventually.

\begin{frame} % {Waxing poetic}
S~Cook: 

\begin{quote}\small % https://www.claymath.org/sites/default/files/pvsnp.pdf
\noindent \ldots{} a practical algorithm for solving an $\NP$-complete problem
\ldots{}
would transform mathematics by allowing a computer to find a formal
proof of any theorem that has a proof of reasonable length, 
since formal proofs can
easily be recognized in polynomial time. 
\ldots{} 
Although the formal proofs may not be initially intelligible
to humans, the problem of finding intelligible proofs would be reduced to that
of finding a recognition algorithm for intelligible proofs. 
Similar remarks apply
to diverse creative human endeavors, such as designing airplane wings, creating
physical theories, or even composing music. 
The question in each case is to what
extent an efficient algorithm for recognizing a good result can be found.
\end{quote}

S~Aaronson: % https://www.scottaaronson.com/blog/?p=122

\begin{quote}\small
If $\P=\NP$, then the world would be a profoundly different place 
than we usually assume it to be. 
There would be no special value in ``creative leaps,'' no fundamental 
gap between solving a problem and recognizing the solution once it's found. 
Everyone who could appreciate a symphony would be Mozart; everyone 
who could follow a step-by-step argument would be Gauss; everyone who 
could recognize a good investment strategy would be Warren Buffett.
\end{quote}

A Wigderson: % http://www.math.ias.edu/~avi/PUBLICATIONS/MYPAPERS/AW09/AW09.pdf

\begin{quote}\small
The seemingly abstract, philosophical question: 
\textit{Can creativity be automated?} in its concrete, mathematical 
form: \textit{Does $\P = \NP$?},
emerges as a central challenge of science. And the basic notions of
space and time, studied as resources of efficient computation, emerge
as key objects of study to solving this mystery, just like their physical
counterparts hold the key to understanding the laws of nature.
\end{quote}

\begin{frame}{Guessing whether $\P=\NP$: the dominant view}
\ExecuteMetaData[\catchfilefn]{discussion:PvsNPiii}

\pause
% credit: https://math.stackexchange.com/a/2547613/12012
Related reasoning is that there are thousands of $\NP$ complete problems 
spread across disciplines such as combinatorics, 
number theory, geometry, economics and finance, biology, operations research, 
etc. 
These problems differ wildly and bear no obvious relation to each other, 
besides the fact that they are all $\NP$ complete. 
Because they are $\NP$ complete, they are mutually inter-reducible,
so a polytime solution to one would give a polytime solution to all.
No one has been able to find a polytime algorithm to solve any of them.
This is strong empirical evidence\Dash not proof, to be sure, but still strong
evidence\Dash that $\P\neq\NP$.
\end{frame}
